# Format reference for the ax-task default-branch cache.
#
# The live cache is NOT this file: ax-task reads and appends to
# <notes repo>/.state/default-branches.txt, creating it on first use.
# One line per repo: "<repo> <branch>". Lines starting with "#" are ignored.
#
# Example:
# api-gateway main
# web-app main
# mobile-app develop
# billing-service master
